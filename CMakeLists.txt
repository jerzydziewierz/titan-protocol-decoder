cmake_minimum_required(VERSION 3.16)
project(titan_learned LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

add_library(titan STATIC src/titan.cpp)
target_include_directories(titan PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/src")

add_executable(recode src/main.cpp)
target_link_libraries(recode PRIVATE titan)

if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
    target_compile_options(titan PRIVATE -Wall -Wextra -Wpedantic)
    target_compile_options(recode PRIVATE -Wall -Wextra -Wpedantic)
endif()

# Tests exercise the unchanged library; failed assertions are reported independently.
include(CTest)
if(BUILD_TESTING)
    add_executable(titan-tests tests/tests.cpp)
    target_link_libraries(titan-tests PRIVATE titan)
    add_test(NAME learned-decoder COMMAND titan-tests)
    set_tests_properties(learned-decoder PROPERTIES TIMEOUT 30)
endif()
if(BUILD_TESTING AND CMAKE_SYSTEM_NAME STREQUAL "Linux")
    add_executable(titan-no-heap tests/no_heap.cpp)
    target_link_libraries(titan-no-heap PRIVATE titan)
    target_link_options(titan-no-heap PRIVATE -Wl,--wrap=malloc,--wrap=calloc,--wrap=realloc,--wrap=free)
    add_test(NAME learned-no-heap COMMAND titan-no-heap)
    set_tests_properties(learned-no-heap PROPERTIES TIMEOUT 30)
endif()
if(BUILD_TESTING)
    add_executable(titan-resource-probe tests/resource_probe.cpp)
    target_link_libraries(titan-resource-probe PRIVATE titan)
endif()
